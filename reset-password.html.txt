  <link rel="stylesheet" href="assets/styles.css">
  <script src="https://cdn.jsdelivr.net/npm/@supabase/supabase-js@2"></script>
</head>

<body class="admin-body">

  <div class="login-card">
    <h1>Choose New Password</h1>

    <p>
      Enter the new password you want to use for your GMAX Admin account.
    </p>

    <form id="password-form" class="form-grid">

      <div class="field full">
        <label>New password</label>
        <input
          type="password"
          id="password"
          minlength="8"
          required
        >
      </div>

      <div class="field full">
        <label>Confirm new password</label>
        <input
          type="password"
          id="confirm-password"
          minlength="8"
          required
        >
      </div>

      <div class="field full">
        <button class="btn btn-primary" type="submit">
          Save New Password
        </button>
      </div>

    </form>

    <div id="message" style="margin-top:15px;"></div>

  </div>

  <script src="assets/config.js"></script>

  <script>
    const client = window.supabase.createClient(
      window.GMAX_CONFIG.SUPABASE_URL,
      window.GMAX_CONFIG.SUPABASE_ANON_KEY
    );

    const form = document.getElementById('password-form');
    const message = document.getElementById('message');

    form.addEventListener('submit', async function (event) {
      event.preventDefault();

      const password =
        document.getElementById('password').value;

      const confirmPassword =
        document.getElementById('confirm-password').value;

      if (password !== confirmPassword) {
        message.textContent = 'The two passwords do not match.';
        return;
      }

      message.textContent = 'Saving new password...';

      const { error } = await client.auth.updateUser({
        password: password
      });

      if (error) {
        message.textContent = 'Error: ' + error.message;
        return;
      }

      message.innerHTML =
        '<strong>Password changed successfully.</strong><br>You can now return to the GMAX Admin login.';

      setTimeout(function () {
        window.location.href = '/admin';
      }, 2500);
    });
  </script>
